use super::*;
use std::time::Instant;
struct CurlFetcher(Instant);
impl HttpFetcher for CurlFetcher {
    fn fetch(&self, url: &str) -> Result<RawFetchResponse, String> {
        let t = Instant::now();
        eprintln!("[T+{:>6}ms] FETCH start {}", self.0.elapsed().as_millis(), url);
        let out = std::process::Command::new("curl").args(["-s", "-m", "60", "-A", "praetorial-resolver/0.1", "-w", "\n%{http_code} %{content_type}", url]).output().map_err(|e| e.to_string())?;
        let s = out.stdout; let i = s.iter().rposition(|b| *b == b'\n').unwrap();
        let tail = String::from_utf8_lossy(&s[i + 1..]).to_string(); let mut p = tail.splitn(2, ' ');
        let code: u16 = p.next().unwrap().parse().unwrap_or(0); let ct = p.next().unwrap_or("").to_string();
        eprintln!("[T+{:>6}ms] FETCH done  {}ms status={} bytes={}", self.0.elapsed().as_millis(), t.elapsed().as_millis(), code, i);
        Ok(RawFetchResponse { final_url: url.into(), redirect_chain: vec![], content_type: ct, body: s[..i].to_vec(), status_code: code })
    }
}
